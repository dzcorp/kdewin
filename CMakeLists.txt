cmake_minimum_required(VERSION 3.24)
project(lumen VERSION 0.1.0 LANGUAGES CXX)

set(QT_MIN_VERSION "6.9.0")
set(KF_MIN_VERSION "6.20.0")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(ECM ${KF_MIN_VERSION} REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})

include(KDEInstallDirs)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
include(FeatureSummary)

set(LUMEN_GEN_DIR "${CMAKE_SOURCE_DIR}/build/gen/cpp" CACHE PATH "Directory containing the generated lumentokens.h")
if(NOT EXISTS "${LUMEN_GEN_DIR}/lumentokens.h")
    message(FATAL_ERROR "${LUMEN_GEN_DIR}/lumentokens.h not found: run `lumen gen` first")
endif()
get_filename_component(LUMEN_GEN_ROOT "${LUMEN_GEN_DIR}" DIRECTORY)

option(LUMEN_STYLE "Build the Lumen Qt widget style" ON)
option(LUMEN_DECORATION "Build the Lumen window decoration" ON)
option(LUMEN_KWIN "Build the Lumen KWin materials effect" ON)
option(LUMEN_TRAY "Build the Lumen tray (Plasma containment)" ON)
option(LUMEN_TASKS "Build the Lumen taskbar apps applet" ON)
option(LUMEN_NOTIFIER "Build the Lumen notifications applet (server and toasts)" ON)
option(LUMEN_KCM "Build the Lumen System Settings module" ON)
option(LUMEN_NOTIFICATIONS "Build the persistent notification history" ON)
option(LUMEN_DOLPHIN "Build the Lumen Dolphin fork" ON)
option(LUMEN_BENCH "Build benchmarks" OFF)
option(LUMEN_ICONS "Install the windows-modern icon theme" ON)

find_package(Qt6 ${QT_MIN_VERSION} REQUIRED COMPONENTS Core Gui Widgets DBus)
find_package(KF6 ${KF_MIN_VERSION} REQUIRED COMPONENTS ColorScheme Config CoreAddons GuiAddons WindowSystem)

add_subdirectory(common)
add_subdirectory(qml-settings)
if(LUMEN_STYLE)
    add_subdirectory(style)
endif()
if(LUMEN_DECORATION)
    add_subdirectory(decoration)
endif()
if(LUMEN_KWIN)
    add_subdirectory(kwin/effects/materials)
endif()
if(LUMEN_TRAY)
    add_subdirectory(plasma/tray)
endif()
if(LUMEN_TASKS)
    add_subdirectory(plasma/tasks)
endif()
if(LUMEN_NOTIFIER)
    add_subdirectory(plasma/notifier)
endif()
if(LUMEN_KCM)
    add_subdirectory(kcm)
endif()
if(LUMEN_NOTIFICATIONS)
    add_subdirectory(plasma/notifystore)
endif()
if(LUMEN_DOLPHIN)
    add_subdirectory(apps/dolphin)
endif()
if(LUMEN_BENCH)
    add_subdirectory(bench)
endif()
if(LUMEN_ICONS)
    # Third-party (GPL-3.0, see icons/UPSTREAM); Lumen does not select it, the icon theme stays the user's choice.
    install(DIRECTORY icons/windows-modern DESTINATION ${KDE_INSTALL_ICONDIR})
endif()

# Catalogs: po/<lang>/<domain>.po, filled by `lumen i18n`.
find_package(KF6I18n ${KF_MIN_VERSION} REQUIRED)
ki18n_install(po)

feature_summary(WHAT ALL FATAL_ON_MISSING_REQUIRED_PACKAGES)
